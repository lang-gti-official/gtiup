function(manager_target target)
  add_executable(${target} src/main.cpp)
  target_include_directories(${target} PRIVATE "${CMAKE_BINARY_DIR}/dependencies/include" "${libarchive_SOURCE_DIR}/libarchive")
  if(ARGC GREATER 1)
    target_compile_definitions(${target} PRIVATE GTIUP_VERSION="${ARGV1}")
  else()
    target_compile_definitions(${target} PRIVATE GTIUP_VERSION="${PROJECT_VERSION}")
  endif()
  target_link_libraries(${target} PRIVATE ${GTIUP_CURL} archive_static ZLIB::ZLIB)
  if(NOT APPLE)
    target_link_libraries(${target} PRIVATE OpenSSL::Crypto)
  endif()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  if(GTIUP_STATIC)
    target_link_options(${target} PRIVATE -static)
  endif()
  if(GTIUP_SANITIZERS)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
